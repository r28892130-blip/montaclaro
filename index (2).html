function simulateRoulette(trials) {
  let wins = 0;
  for (let i = 0; i < trials; i++) {
    const r = Math.floor(Math.random() * 37);
    if (r === 7) wins++; // ставка на 7
  }
  return wins / trials;
}

function simulateCoinFlips(trials) {
  let heads = 0;
  for (let i = 0; i < trials; i++) {
    // random.choice(["Орёл","Решка"])
    if (Math.random() < 0.5) heads++;
  }
  return heads / trials;
}
